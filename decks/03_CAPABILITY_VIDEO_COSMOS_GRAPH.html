// Canvas Euler Simulation Step (60 FPS)
function simulateStep(nodes: CosmosVideoNode[], damping = 0.92) {
  // 1. Inter-node mutual repulsion (Coulomb / Gravitation)
  for (let i = 0; i < nodes.length; i++) {
    for (let j = i + 1; j < nodes.length; j++) {
      const dx = nodes[j].x - nodes[i].x;
      const dy = nodes[j].y - nodes[i].y;
      const dist = Math.sqrt(dx * dx + dy * dy) || 1;
      const force = (nodes[i].mass * nodes[j].mass) / (dist * dist);
      
      nodes[i].vx -= (dx / dist) * force;
      nodes[i].vy -= (dy / dist) * force;
      nodes[j].vx += (dx / dist) * force;
      nodes[j].vy += (dy / dist) * force;
    }

    // 2. Apply velocity damping & position update
    nodes[i].vx *= damping;
    nodes[i].vy *= damping;
    nodes[i].x += nodes[i].vx;
    nodes[i].y += nodes[i].vy;
  }
}
