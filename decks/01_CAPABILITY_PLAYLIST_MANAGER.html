// Ingesting new clips without overwriting user notes
function mergeIncomingClips(existing, incoming) {
  const map = new Map(existing.map(c => [c.id, c]));

  return incoming.map(inc => {
    const old = map.get(inc.id);
    if (old) {
      return {
        ...inc,
        notes: old.notes,                     // Preserved
        userQuestions: old.userQuestions,     // Preserved
        status: old.status || inc.status,     // Preserved
        voiceReflections: old.voiceReflections // Preserved
      };
    }
    return inc; // Freshly added clip
  });
}
